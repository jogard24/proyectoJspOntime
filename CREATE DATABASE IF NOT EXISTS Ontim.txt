-- 1. Tabla Central de Usuarios
CREATE TABLE usuario (
    id INT AUTO_INCREMENT PRIMARY KEY,
    documento_identidad VARCHAR(30) UNIQUE NOT NULL,
    nombre VARCHAR(100) NOT NULL, 
    direccion VARCHAR(100) NOT NULL,
    estado enum ('activo', 'inactivo') not null,
    tipo_sangre ENUM('a+','a-', 'b+', 'b-', 'o+','o-','ab+','ab-'),
    fotoPerfil_url VARCHAR(255) NOT NULL,
	fecha_registro TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    usuario_modificador_id INT, -- auditoria --
    fecha_modificacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_modificador_id) REFERENCES usuario(id));

-- 2. Tabla de Roles
CREATE TABLE roles (
    id INT AUTO_INCREMENT PRIMARY KEY,
    Rol ENUM('administrador','empleado','contador') UNIQUE NOT NULL
);



-- 3. Tabla de Credenciales (Se eliminó la restricción UNIQUE en rol_id para permitir varios contadores/empleados)
CREATE TABLE credenciales (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT,
    usuario VARCHAR(100) UNIQUE,
    clave VARCHAR(50),
    rol_id INT,
    activo BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (usuario_id) REFERENCES usuario(id),
    FOREIGN KEY (rol_id) REFERENCES roles(id)
);

-- 4. Tabla de Jornadas Laborales (Debe ir ANTES de asistencia)
CREATE TABLE jornadaLaboral(
    id INT PRIMARY KEY,
    nombrejornada VARCHAR(50) UNIQUE NOT NULL,
    hora_entrada TIME, 
    hora_salida TIME
);

--  Tabla de Asistencia
CREATE TABLE asistencia (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT,
    tipo_evento ENUM('entrada', 'salida'),
    fecha_hora DATETIME DEFAULT CURRENT_TIMESTAMP,
    observacion TEXT,
    jornada_id INT NOT NULL,
    FOREIGN KEY (usuario_id) REFERENCES usuario(id),
    FOREIGN KEY (jornada_id) REFERENCES jornadaLaboral(id)
);


-- 6. Tabla de Contratos (Para cálculos de salarios de nómina)
CREATE TABLE contrato (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT NOT NULL,
    tipo_contrato VARCHAR(100) NOT NULL,
    cargo VARCHAR(100) NOT NULL,
    salario_base DECIMAL(10,2) NOT NULL,
    jornada_id INT,
    FOREIGN KEY (usuario_id) REFERENCES usuario(id),
    FOREIGN KEY (jornada_id) REFERENCES jornadaLaboral(id)
);

CREATE TABLE permiso_laboral (
id INT AUTO_INCREMENT PRIMARY KEY,
permiso_id INT NOT NULL,
tipo_permiso varchar (150) NOT NULL,
fecha_asignacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
fecha_inicio date ,
fecha_fin date ,
estado BOOLEAN DEFAULT FALSE,
foreign key (permiso_id) references roles(id));

-- tiempo y eventos
CREATE TABLE periodo_nomina (
    id INT AUTO_INCREMENT PRIMARY KEY,
    fecha_inicio DATE NOT NULL,
    fecha_fin DATE NOT NULL,
    estado ENUM('abierto', 'cerrado') DEFAULT 'abierto'
);
    
    
-- Nómina (Conectando el tiempo con el periodo)
CREATE TABLE nomina (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT NOT NULL,
    periodo_id INT NOT NULL,
    salario_base_periodo DECIMAL(10,2), -- Captura el salario al momento de pagar
    total_horas_extras DECIMAL(5,2) DEFAULT 0,
    total_neto DECIMAL(10,2),
    FOREIGN KEY (usuario_id) REFERENCES usuario(id),
    FOREIGN KEY (periodo_id) REFERENCES periodo_nomina(id)
);

-- Tabla de conceptos (Define el extra o abono)
CREATE TABLE concepto_nomina (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL, -- Ej: "Hora Extra Diurna", "Llegada Tarde", "Bono Puntualidad"
    tipo ENUM('extra', 'descuento') NOT NULL,
    factor_calculo DECIMAL(5,2) DEFAULT 1.0 -- Ejemplo: 1.25 para extras
);

-- Tabla de detalle (Aquí guardas los movimientos del periodo)
CREATE TABLE detalle_nomina (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nomina_id INT NOT NULL, -- Conexión a la tabla que creamos antes
    concepto_id INT NOT NULL,
    valor DECIMAL(10,2) NOT NULL,
    fecha_registro TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    observacion TEXT, -- Ej: "Llegada tarde el día 05/06/2026"
    FOREIGN KEY (nomina_id) REFERENCES nomina(id),
    FOREIGN KEY (concepto_id) REFERENCES concepto_nomina(id)
);
